INSERT : INSERT INTO 테이블명 (필드명1, 필드명2,...) VALUES (값1, 값2,...);
--         INSERT INTO 테이블명 VALUES (값1,값2,값3,...값N);
UPDATE : UPDATE 테이블명 SET 필드명1=값1, [필드명2=값2,... 필드명n=값n] [WHERE 조건]; []생략가능
DELETE : DELETE FROM 테이블명 [WHERE 조건];
